排序算法速查笔记

一、排序算法清单

算法平均时间最坏时间空间稳定性应用场景

二、各算法思路与代码实现

1. 快速排序 (Lomuto 分区)

思路：随机选 pivot 换到最右，用 i 标记第一个 > pivot 的位置，j 扫描，最后 pivot 归位并递归。

#include <vector>
#include <cstdlib>
#include <ctime>
using namespace std;

int partition(vector<int>& nums, int left, int right) {
    // 随机 pivot 换到 right，避免有序退化
    int randIdx = left + rand() % (right - left + 1);
    swap(nums[randIdx], nums[right]);
    int pivot = nums[right];
    int i = left;                       // i 指向第一个 > pivot 的位置
    for (int j = left; j < right; ++j) {
        if (nums[j] <= pivot) {
            swap(nums[i], nums[j]);
            ++i;
        }
    }
    // 交换前：i 是“第一个大于 pivot 的位置”。
	// 交换后：i 变成了pivot 最终所在的位置”
    swap(nums[i], nums[right]);         // pivot 归位
    return i;
}

void quickSort(vector<int>& nums, int left, int right) {
    if (left >= right) return;
    int p = partition(nums, left, right);
    quickSort(nums, left, p - 1);
    quickSort(nums, p + 1, right);
}

2. 三路快排（处理大量重复元素）

思路：用 lt、i、gt 将数组分成 <、=、> 三部分，等于区跳过递归。
（右端 pivot 版）

void quickSort3Way(vector<int>& nums, int left, int right) {
    if (left >= right) return;

    // 1. 随机选一个数和最右边的数交换
    int randIdx = left + rand() % (right - left + 1);
    swap(nums[randIdx], nums[right]);
    int pivot = nums[right];

    // 2. 准备三个位置标记
    int lt = left;   // 这个标记的左边都是比 pivot 小的
    int i  = left;   // 这个标记指向现在要检查的数
    int gt = right;  // 这个标记的右边都是比 pivot 大的

    // 3. 只要 i 还没碰到 gt，就一直检查
    while (i < gt) {
        if (nums[i] < pivot) {
            swap(nums[lt], nums[i]); // 把小的换到 lt 位置
            lt++;                    // lt 右移一格
            i++;                     // i 右移一格
        }
        else if (nums[i] > pivot) {
	        // 先 `gt--` 是为了“腾出一个空位”来接收大于 pivot 的元素，同时确保换过来的元素能被 `i` 再次检查，且 pivot 始终安全待在 `right` 位置**
            gt--;                    // gt 左移一格
            swap(nums[i], nums[gt]); // 把大的换到 gt 新位置
            // i 不动！因为换过来的数还没检查
        }
        else {
            i++;                     // 相等，直接看下一个
        }
    }

    // 4. 把 pivot 放到中间相等区的末尾
    swap(nums[gt], nums[right]);

    // 5. 对左边小的部分和右边大的部分继续排序，中间相等区不管了
    quickSort3Way(nums, left, lt - 1);
    quickSort3Way(nums, gt + 1, right);
}

4. 插入排序

思路：将元素插入已排序区合适位置，已排序元素边后移边比较。

void insertionSort(vector<int>& nums) {
    int n = nums.size();
    for (int i = 1; i < n; ++i) {
        int key = nums[i], j = i - 1;
        while (j >= 0 && nums[j] > key) {
            nums[j + 1] = nums[j];
            --j;
        }
        nums[j + 1] = key;
    }
}

5. 归并排序

思路：递归二分至单元素，再合并有序子数组。

void merge(vector<int>& nums, int left, int mid, int right) {
    vector<int> tmp(right - left + 1);
    int i = left, j = mid + 1, k = 0;
    while (i <= mid && j <= right)
        tmp[k++] = (nums[i] <= nums[j]) ? nums[i++] : nums[j++];
    while (i <= mid) tmp[k++] = nums[i++];
    while (j <= right) tmp[k++] = nums[j++];
    for (int p = 0; p < tmp.size(); ++p) nums[left + p] = tmp[p];
}

void mergeSort(vector<int>& nums, int left, int right) {
    if (left >= right) return;
    int mid = left + (right - left) / 2;
    mergeSort(nums, left, mid);
    mergeSort(nums, mid + 1, right);
    merge(nums, left, mid, right);
}

6. 优先队列排序

思路：用大顶堆（默认）全部入队，再倒序出队得到升序。
（也可用小顶堆直接正向填充）

#include <queue>
void priorityQueueSort(vector<int>& nums) {
    priority_queue<int> pq;                     // 大顶堆
    for (int x : nums) pq.push(x);
    for (int i = nums.size() - 1; i >= 0; --i) {
        nums[i] = pq.top();
        pq.pop();
    }
}
