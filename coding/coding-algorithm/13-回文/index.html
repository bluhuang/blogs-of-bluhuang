核心区分：什么时候用什么？

只需看一个关键问题：

题目是否需要“频繁判断任意区间 [i, j] 是否为回文”？

情况用什么为什么
只需要找最长的那个（LeetCode 5）中心扩展法不需要预处理所有区间，边扩展边记录最大值即可
只需要统计个数（LeetCode 647）中心扩展法扩展过程中顺便计数，O(1) 空间
需要反复判断不同区间（如回溯、DP 转移中频繁查表）二维 DP 预处理一次性算出所有 dp[i][j]，后续 O(1) 查询

具体判断流程

题目要求什么？
├── 输出一个值（最长子串长度/位置、计数）
│   └── 用「中心扩展法」
│       例题：5. 最长回文子串、647. 回文子串计数
│
├── 输出所有方案（所有分割方式）
│   └── 用「回溯 + 二维 DP 预处理」
│       例题：131. 分割回文串
│
├── 求最少分割次数
│   └── 用「一维 DP + 二维 DP 预处理」
│       例题：132. 分割回文串 II
│
└── 子序列问题（不连续）
    └── 用「区间 DP」（不是中心扩展，因为子序列不连续）
        例题：516. 最长回文子序列

回文（Palindrome）串题型完整笔记

一、题型识别与核心特征

LeetCode 中所有回文串问题都围绕一个核心定义：

回文串：正着读和倒着读一样的字符串。

问题类型通常分为三类：

  1. 验证/计数型：判断是否是回文、统计回文子串数量。
  2. 构造/分割型：分割出所有回文子串（回溯）、最少分割次数（DP）。
  3. 最长/极值型：找最长的回文子串、最长回文子序列。
题型代表题目核心解法
最长回文子串5. Longest Palindromic Substring中心扩展法
回文子串计数647. Palindromic Substrings中心扩展法
验证回文串125. Valid Palindrome两端双指针
验证回文串 II680. Valid Palindrome II两端双指针 + 贪心
分割回文串131. Palindrome Partitioning回溯 + 二维DP预处理
分割回文串 II132. Palindrome Partitioning II一维DP + 二维DP预处理
最长回文子序列516. Longest Palindromic Subsequence区间DP
回文对336. Palindrome Pairs哈希表 + 切割判断

二、核心抉择：什么时候用中心扩展法，什么时候用二维DP？

这是解决回文题最重要的判断依据。

决策口诀

“只找一个用扩展，频繁查表用 DP，子序列必用区间 DP。”

具体判断流程

题目要求什么？
├── 输出一个值（最长/计数），不需要反复查表
│   └── 用「中心扩展法」，O(n²) 时间，O(1) 空间
│       例题：5. 最长回文子串、647. 回文子串计数
│
├── 输出所有方案（回溯），或求最优解（DP转移中需频繁判断）
│   └── 用「二维 DP 预处理」一次性算出所有区间，后续 O(1) 查询
│       例题：131. 分割回文串（回溯查表）、132. 分割回文串 II（DP查表）
│
└── 子序列问题（不连续）
    └── 用「区间 DP」，不能用中心扩展
        例题：516. 最长回文子序列

为什么这样划分？

方法擅长的事不擅长的事
中心扩展法按顺序扩展、找最长/计数，空间 O(1)无法高效回答“任意区间 [i, j] 是否为回文”
二维 DP 预处理一次性算出所有区间，后续查表 O(1)空间 O(n²)，不适合只需找最长的情况

三、基础：判断子串是否回文的两种方法

1. 双指针中心扩展法（求最长/计数时使用）

int expand(const string &s, int left, int right) {
    while (left >= 0 && right < s.size() && s[left] == s[right]) {
        left--;
        right++;
    }
    return right - left - 1; // 实际回文长度
}

2. 二维DP预处理（需要频繁查表时使用）

int n = s.size();
vector<vector<bool>> dp(n, vector<bool>(n, false));
for (int j = 0; j < n; ++j) {
    for (int i = 0; i <= j; ++i) {
        if (s[i] == s[j] && (j - i <= 2 || dp[i+1][j-1])) {
            dp[i][j] = true;
        }
    }
}

四、核心题深度解析

题目 1：最长回文子串（LeetCode 5）

题目要求：给定一个字符串 s，找到 s 中最长的回文子串。

为什么用中心扩展法：只需找最长的那个，不需要反复判断任意区间。

class Solution {
public:
    string longestPalindrome(string s) {
        int start = 0, maxLen = 1;
        for (int i = 0; i < s.size(); ++i) {
            int len1 = expand(s, i, i);
            int len2 = expand(s, i, i + 1);
            int len = max(len1, len2);
            if (len > maxLen) {
                maxLen = len;
                start = i - (len - 1) / 2;
            }
        }
        return s.substr(start, maxLen);
    }
    
    int expand(const string &s, int left, int right) {
        while (left >= 0 && right < s.size() && s[left] == s[right]) {
            left--;
            right++;
        }
        return right - left - 1;
    }
};

题目 2：回文子串计数（LeetCode 647）

题目要求：统计回文子串的数目。

为什么用中心扩展法：只需计数，扩展过程中顺便计数即可。

class Solution {
public:
    int countSubstrings(string s) {
        int count = 0;
        for (int i = 0; i < s.size(); ++i) {
            count += expandCount(s, i, i);
            count += expandCount(s, i, i + 1);
        }
        return count;
    }
    
    int expandCount(const string &s, int left, int right) {
        int cnt = 0;
        while (left >= 0 && right < s.size() && s[left] == s[right]) {
            cnt++;
            left--;
            right++;
        }
        return cnt;
    }
};

题目 3：验证回文串（LeetCode 125）

题目要求：验证字符串是否是回文，只考虑字母数字，忽略大小写。

为什么用两端双指针：给定了完整字符串，只需判断一次。

class Solution {
public:
    bool isPalindrome(string s) {
        int left = 0, right = s.size() - 1;
        while (left < right) {
            while (left < right && !isalnum(s[left])) left++;
            while (left < right && !isalnum(s[right])) right--;
            if (tolower(s[left]) != tolower(s[right])) return false;
            left++;
            right--;
        }
        return true;
    }
};

题目 4：验证回文串 II（LeetCode 680）

题目要求：最多删除一个字符，判断能否成为回文。

为什么用两端双指针：与125类似，但遇到不等时尝试跳过一边。

class Solution {
public:
    bool validPalindrome(string s) {
        int left = 0, right = s.size() - 1;
        while (left < right) {
            if (s[left] != s[right]) {
                return isPalindrome(s, left + 1, right) || 
                       isPalindrome(s, left, right - 1);
            }
            left++;
            right--;
        }
        return true;
    }
    
    bool isPalindrome(const string &s, int left, int right) {
        while (left < right) {
            if (s[left] != s[right]) return false;
            left++;
            right--;
        }
        return true;
    }
};

题目 5：分割回文串（LeetCode 131）

题目要求：返回所有可能的分割方案，使每个子串都是回文。

为什么用二维 DP 预处理：回溯过程中需要反复判断不同区间是否为回文，预处理后 O(1) 查询。

class Solution {
public:
    vector<vector<string>> partition(string s) {
        int n = s.size();
        // 二维 DP 预处理
        vector<vector<bool>> dp(n, vector<bool>(n, false));
        for (int j = 0; j < n; ++j)
            for (int i = 0; i <= j; ++i)
                if (s[i] == s[j] && (j - i <= 2 || dp[i+1][j-1]))
                    dp[i][j] = true;
        
        vector<vector<string>> res;
        vector<string> path;
        function<void(int)> dfs = [&](int start) {
            if (start == n) {
                res.push_back(path);
                return;
            }
            for (int end = start; end < n; ++end) {
                if (dp[start][end]) {
                    path.push_back(s.substr(start, end - start + 1));
                    dfs(end + 1);
                    path.pop_back();
                }
            }
        };
        dfs(0);
        return res;
    }
};

题目 6：分割回文串 II（LeetCode 132）

题目要求：返回最少分割次数，使每个子串都是回文。

为什么用二维 DP 预处理：一维 DP 的转移过程中需要频繁查表。

class Solution {
public:
    int minCut(string s) {
        int n = s.size();
        // 二维 DP 预处理
        vector<vector<bool>> pal(n, vector<bool>(n, false));
        for (int j = 0; j < n; ++j)
            for (int i = 0; i <= j; ++i)
                if (s[i] == s[j] && (j - i <= 2 || pal[i+1][j-1]))
                    pal[i][j] = true;
        
        // 一维 DP 求最少分割
        vector<int> dp(n + 1, INT_MAX);
        dp[0] = -1;
        for (int i = 1; i <= n; ++i) {
            for (int j = 0; j < i; ++j) {
                if (pal[j][i-1]) {
                    dp[i] = min(dp[i], dp[j] + 1);
                }
            }
        }
        return dp[n];
    }
};

题目 7：最长回文子序列（LeetCode 516）

题目要求：找出最长回文子序列的长度（子序列可以不连续）。

为什么不能用中心扩展法：子序列不连续，必须用区间 DP。

class Solution {
public:
    int longestPalindromeSubseq(string s) {
        int n = s.size();
        vector<vector<int>> dp(n, vector<int>(n, 0));
        for (int i = 0; i < n; ++i) dp[i][i] = 1;
        
        for (int len = 2; len <= n; ++len) {
            for (int i = 0; i + len - 1 < n; ++i) {
                int j = i + len - 1;
                if (s[i] == s[j])
                    dp[i][j] = dp[i+1][j-1] + 2;
                else
                    dp[i][j] = max(dp[i+1][j], dp[i][j-1]);
            }
        }
        return dp[0][n-1];
    }
};

题目 8：回文对（LeetCode 336）

题目要求：找所有索引对，使得 words[i] + words[j] 是回文。

解法：哈希表 + 切割判断（不属于中心扩展 vs DP 的二分问题）。

class Solution {
public:
    vector<vector<int>> palindromePairs(vector<string>& words) {
        unordered_map<string, int> dict;
        for (int i = 0; i < words.size(); ++i) {
            string rev = words[i];
            reverse(rev.begin(), rev.end());
            dict[rev] = i;
        }
        
        vector<vector<int>> res;
        for (int i = 0; i < words.size(); ++i) {
            string word = words[i];
            int n = word.size();
            for (int j = 0; j <= n; ++j) {
                string left = word.substr(0, j);
                string right = word.substr(j);
                if (isPalindrome(left)) {
                    string rev = right; reverse(rev.begin(), rev.end());
                    if (dict.count(rev) && dict[rev] != i)
                        res.push_back({dict[rev], i});
                }
                if (!right.empty() && isPalindrome(right)) {
                    string rev = left; reverse(rev.begin(), rev.end());
                    if (dict.count(rev) && dict[rev] != i)
                        res.push_back({i, dict[rev]});
                }
            }
        }
        return res;
    }
    
    bool isPalindrome(const string &s) {
        int l = 0, r = s.size() - 1;
        while (l < r) if (s[l++] != s[r--]) return false;
        return true;
    }
};

五、总结：一次记住所有抉择

题目类型解法选择一句话原因
找最长子串（5）中心扩展法只找一个，O(1) 空间
统计个数（647）中心扩展法扩展顺便计数
验证是否是回文（125, 680）两端双指针给定完整串，判断一次
输出所有分割方案（131）回溯 + 二维 DP 预处理回溯需频繁查表
求最少分割次数（132）一维 DP + 二维 DP 预处理DP 转移需频繁查表
子序列问题（516）区间 DP不连续，必须区间 DP

终极口诀：

“只找最长扩展它，频繁查表 DP 它，子序列必区间它。”
