    回文（Palindrome）串题型完整笔记#

    一、题型识别与核心特征#

    所有回文串问题（常见于在线编程平台）都围绕一个核心定义：

    回文串：正着读和倒着读一样的字符串。

    问题类型通常分为三类：

    1. 验证/计数型：判断是否是回文、统计回文子串数量。
    2. 构造/分割型：分割出所有回文子串（回溯）、最少分割次数（DP）。
    3. 最长/极值型：找最长的回文子串。

    所有问题的根基都是如何高效判断一个子串是回文。一旦熟练掌握子串回文的判断，所有问题无非是在此基础上遍历、计数、分割或求极值。

    二、常见误区：为什么“从两边向中间验证”不能用来找最长子串？#

    可能会下意识想：

    “只要枚举每一个子串起点和终点，然后用双指针从两边向中间验证它是不是回文，最后记录最长的就行了。”

    这个思路逻辑完全正确，但时间复杂度过高。

    • 枚举所有子串需要 O(n²) 个起点-终点对。
    • 每次验证又是 O(n) 的时间。
    • 总时间复杂度高达 O(n³)，在 n=1000 时基本会超时。

    结论：

    从两边向中间的双指针是验证已知子串的完美方法（用于 125、680 等题），但不适合用来“搜索”最长回文子串。 搜索场景下，使用中心扩展法（O(n²) 时间，O(1) 空间）或动态规划（O(n²) 时间，O(n²) 空间）。

    三、基础：判断子串是否回文的两种方法#

    1. 双指针中心扩展法（推荐，O(1) 空间）#

    从中心向两边扩展，一次遍历即可同时处理奇数和偶数长度。

    // 在 s 上以 left, right 为中心向两边扩展，返回能扩展的最大回文长度
int expandAroundCenter(const string &s, int left, int right) {
    while (left >= 0 && right < s.size() && s[left] == s[right]) {
        left--;
        right++;
    }
    return right - left - 1; // 实际回文长度
}

    2. 二维动态规划（适合一次性判断多个子串）#

    int n = s.size();
vector<vector<bool>> dp(n, vector<bool>(n, false));
int maxLen = 1, start = 0;

for (int j = 0; j < n; ++j) {
    for (int i = 0; i <= j; ++i) {
        if (s[i] == s[j] && (j - i <= 2 || dp[i+1][j-1])) {
            dp[i][j] = true;
            // 更新 maxLen 和 start
        }
    }
}

    四、必考题深度解析#

    题目 1：最长回文子串（第5题）#

    题目要求：给定一个字符串 s，找到 s 中最长的回文子串。如果存在多个，返回任意一个。

    思路：遍历每个字符，以它为中心（奇数长度）和以它及下一个字符为中心（偶数长度）向两边扩展，记录能扩展到的最长回文子串的起点和长度。

    代码：

    class Solution {
public:
    string longestPalindrome(string s) {
        if (s.empty()) return "";
        int start = 0, maxLen = 1;
        for (int i = 0; i < s.size(); ++i) {
            int len1 = expand(s, i, i);     // 奇数长度中心
            int len2 = expand(s, i, i + 1); // 偶数长度中心
            int len = max(len1, len2);
            if (len > maxLen) {
                maxLen = len;
                start = i - (len - 1) / 2;
            }
        }
        return s.substr(start, maxLen);
    }
    
    int expand(const string &s, int left, int right) {
        while (left >= 0 && right < s.size() && s[left] == s[right]) {
            left--;
            right++;
        }
        return right - left - 1;
    }
};

    题目 2：回文子串计数（第647题）#

    题目要求：给定一个字符串 s，统计并返回这个字符串中回文子串的数目。

    思路：与上一题完全相同，只是将“记录最长长度”改为“每扩展一步，计数器加一”。

    代码：

    class Solution {
public:
    int countSubstrings(string s) {
        int count = 0;
        for (int i = 0; i < s.size(); ++i) {
            count += expandCount(s, i, i);
            count += expandCount(s, i, i + 1);
        }
        return count;
    }
    
    int expandCount(const string &s, int left, int right) {
        int cnt = 0;
        while (left >= 0 && right < s.size() && s[left] == s[right]) {
            cnt++;
            left--;
            right++;
        }
        return cnt;
    }
};

    题目 3：验证回文串（第125题）#

    题目要求：给定一个字符串 s，验证它是否是回文串，只考虑字母和数字字符，忽略大小写。

    思路：双指针从两端向中间扫描，跳过非字母数字字符，比较时统一转成小写。

    代码：

    class Solution {
public:
    bool isPalindrome(string s) {
        int left = 0, right = s.size() - 1;
        while (left < right) {
            while (left < right && !isalnum(s[left])) left++;
            while (left < right && !isalnum(s[right])) right--;
            if (tolower(s[left]) != tolower(s[right])) return false;
            left++;
            right--;
        }
        return true;
    }
};

    题目 4：验证回文串 II（第680题）#

    题目要求：给定一个非空字符串 s，最多删除一个字符，判断是否能成为回文串。

    思路：先用双指针从两端向中间比较，遇到不相等时，分别尝试跳过左边或跳过右边，判断剩下的子串是否为回文。

    代码：

    class Solution {
public:
    bool validPalindrome(string s) {
        int left = 0, right = s.size() - 1;
        while (left < right) {
            if (s[left] != s[right]) {
                return isPalindrome(s, left + 1, right) || 
                       isPalindrome(s, left, right - 1);
            }
            left++;
            right--;
        }
        return true;
    }
    
    bool isPalindrome(const string &s, int left, int right) {
        while (left < right) {
            if (s[left] != s[right]) return false;
            left++;
            right--;
        }
        return true;
    }
};

    题目 5：分割回文串（第131题）#

    题目要求#

    给定一个字符串 s，将 s 分割成一些子串，使每个子串都是回文串。返回所有可能的分割方案。

    详细思路#

    这个问题需要输出所有合法的切分方式，自然想到回溯（DFS）。

    想象我们站在字符串的某个位置 start，我们需要决定：从哪里切一刀？切下去的前提是当前这段子串必须是回文。如果 s[start..end] 是回文，我们就把它放入当前路径，然后递归处理剩余部分 end+1。当 start 走到字符串末尾时，就把整条路径加入结果。

    问题是：在回溯过程中，需要反复检查 s[start..end] 是不是回文。如果每次都用双指针验证，重复计算太多。因此先用二维 DP 预处理出任意区间 [i,j] 是否为回文。之后回溯时，判断就是 O(1) 的了。

    DP 预处理：

    • dp[i][j] 表示 s[i..j] 是否为回文。
    • 转移：s[i] == s[j] 且（区间长度 ≤2 或 dp[i+1][j-1] 为真）。
    • 填表顺序：按子串长度从小到大或右边界 j 递增。

    代码#

    class Solution {
public:
    vector<vector<string>> partition(string s) {
        int n = s.size();
        // 1. DP 预处理所有子串的回文性
        vector<vector<bool>> dp(n, vector<bool>(n, false));
        for (int j = 0; j < n; ++j)
            for (int i = 0; i <= j; ++i)
                if (s[i] == s[j] && (j - i <= 2 || dp[i+1][j-1]))
                    dp[i][j] = true;
        
        // 2. 回溯收集所有分割方案
        vector<vector<string>> res;
        vector<string> path;
        
        function<void(int)> dfs = [&](int start) {
            if (start == n) {
                res.push_back(path);
                return;
            }
            for (int end = start; end < n; ++end) {
                if (dp[start][end]) {
                    path.push_back(s.substr(start, end - start + 1));
                    dfs(end + 1);
                    path.pop_back();
                }
            }
        };
        
        dfs(0);
        return res;
    }
};
